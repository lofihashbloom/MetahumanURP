cmake_minimum_required(VERSION 3.15)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(MetaHumanRigLogic LANGUAGES CXX)
if(NOT OPEN_RIGLOGIC_SOURCE)
    message(FATAL_ERROR "Set OPEN_RIGLOGIC_SOURCE to the pinned Epic OpenRigLogic 5.8 checkout")
endif()
set(RL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RL_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(RL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RL_BUILD_PIC ON CACHE BOOL "" FORCE)
set(RL_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory("${OPEN_RIGLOGIC_SOURCE}" riglogic)
add_library(metahuman_riglogic SHARED bridge.cpp)
target_compile_features(metahuman_riglogic PRIVATE cxx_std_17)
target_link_libraries(metahuman_riglogic PRIVATE RigLogic::riglogic4)
target_include_directories(metahuman_riglogic PRIVATE "${OPEN_RIGLOGIC_SOURCE}/include")
set_target_properties(metahuman_riglogic PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
